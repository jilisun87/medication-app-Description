==========================================
第一步：重置 MySQL root 密码
==========================================

# 查看 MySQL 状态
sudo systemctl status mysql | head -5

# 尝试空密码（腾讯云默认有时是空密码）
sudo mysql -e "SELECT 1" 2>&1 && echo "空密码OK" || echo "需要密码"

# 如果上面失败，试旧密码（填入你知道的旧密码）
sudo mysql -p'你的旧密码' -e "SELECT 1" 2>&1 && echo "旧密码OK"

# 连接成功后，重置为统一密码
sudo mysql -e "ALTER USER 'root'@'localhost' IDENTIFIED WITH mysql_native_password BY 'MedicationDB2024Safe'; FLUSH PRIVILEGES;"
echo "MySQL密码已重置"

# 验证
sudo mysql -u root -p'MedicationDB2024Safe' -e "SELECT '连接成功' as test"

USE medication_db;

CREATE TABLE IF NOT EXISTS pharmacist_accounts (
    id VARCHAR(36) PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    name VARCHAR(50) NOT NULL,
    dept VARCHAR(100) DEFAULT NULL,
    hospital VARCHAR(200) DEFAULT NULL,
    phone VARCHAR(20) DEFAULT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_username (username)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

==========================================
第三步：下载 v2.0 后端 app.py
==========================================

# 确认路径存在
ls /var/www/medication-api/

# 下载 v2.0 app.py（commit 8ca275d）
sudo wget -qO /var/www/medication-api/app.py \
  "https://raw.githubusercontent.com/jilisun87/medication-app-Description/8ca275d/backend/app.py"

echo "下载字节数:"
sudo wc -c /var/www/medication-api/app.py

==========================================
第四步：重启后端服务
==========================================

# 杀掉旧进程
sudo pkill -f "uvicorn app:app" 2>/dev/null
sleep 2

# 进入虚拟环境，重启
cd /var/www/medication-api
source venv/bin/activate
nohup uvicorn app:app --host 0.0.0.0 --port 8000 > /tmp/api.log 2>&1 &
sleep 3

# 测试
curl -s http://localhost:8000/health
echo ""

==========================================
第五步：下载前端文件
